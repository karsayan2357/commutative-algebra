\begin{convention}
By a ring, we mean a commutative ring with $1$. Also, we exclude the zero ring.
\end{convention}

\begin{enumerate}

\item Let $I$ be an ideal of $R$ and $\rad(I) := \{a \in R \mid a^n \in I \text{ for some } n \geq 1\}$. Prove the following:
\begin{enumerate}[label=(\alph*)]
    \item $\rad(\rad(I)) = \rad(I)$

    \begin{proof}[Solution]
    The containment $\rad(I) \subseteq \rad(\rad(I))$ is immediate, since $I \subseteq \rad(I)$ and $\rad(-)$ is monotone.

    Conversely, let $x \in \rad(\rad(I))$. Then $x^n \in \rad(I)$ for some $n$, so $(x^n)^m = x^{mn} \in I$ for some $m$. Hence $x \in \rad(I)$.
    \end{proof}

    \item $\rad(IJ) = \rad(I \cap J) = \rad(I) \cap \rad(J)$

    \begin{proof}[Solution]
    We show $\rad(I)\cap\rad(J) \subseteq \rad(IJ) \subseteq \rad(I\cap J) \subseteq \rad(I)\cap\rad(J)$, giving all three sets equal.
    \begin{itemize}
        \item $\rad(I)\cap\rad(J) \subseteq \rad(IJ)$: if $x\in\rad(I)\cap\rad(J)$, then $x^n\in I$ and $x^m\in J$ for some $n,m$, so $x^{n+m}=x^nx^m\in IJ$, i.e.\ $x\in\rad(IJ)$.
        \item $\rad(IJ)\subseteq\rad(I\cap J)$: since $IJ\subseteq I\cap J$, this follows from monotonicity of $\rad$.
        \item $\rad(I\cap J)\subseteq \rad(I)\cap\rad(J)$: since $I\cap J\subseteq I$ and $I\cap J\subseteq J$, monotonicity gives $\rad(I\cap J)\subseteq\rad(I)$ and $\rad(I\cap J)\subseteq\rad(J)$.
    \end{itemize}
    \end{proof}

    \item $\rad(I) = R$ if and only if $I = R$

    \begin{proof}[Solution]
    ($\Rightarrow$) If $\rad(I)=R$, then for every $a\in R$ there is $n\in\bbn$ with $a^n\in I$. Taking $a=1$: $1^n=1\in I$, so $I=R$.

    ($\Leftarrow$) Trivial: if $I=R$ then $\rad(I)=R$ as well.
    \end{proof}

    \item $\rad(I + J) = \rad(\rad(I) + \rad(J))$

    \begin{proof}[Solution]
    ($\subseteq$) Let $x\in\rad(I+J)$. Then $x^k = u+v$ for some $k\in\bbn$, $u\in I$, $v\in J$. Since $u\in I\subseteq\rad(I)$ and $v\in J\subseteq\rad(J)$, $x^k\in\rad(I)+\rad(J)$, so $x\in\rad(\rad(I)+\rad(J))$.

    ($\supseteq$) Since $I \subseteq I+J$, monotonicity gives $\rad(I)\subseteq\rad(I+J)$; similarly $\rad(J)\subseteq\rad(I+J)$. As $\rad(I+J)$ is an ideal, $\rad(I)+\rad(J)\subseteq\rad(I+J)$. Applying $\rad(-)$ and using part (a),
    \[
    \rad(\rad(I)+\rad(J)) \subseteq \rad(\rad(I+J)) = \rad(I+J).
    \]
    \end{proof}

    \item If $P \in \Spec(R)$, then $\rad(P^n) = P$ for all $n > 0$

    \begin{proof}[Solution]
    First, $\rad(P) = P$: the containment $P\subseteq\rad(P)$ always holds, and $\rad(P)\subseteq P$ follows since $P$ is prime (if $x^n\in P$, primality forces $x\in P$).

    Now induct on $n$. The base case $n=1$ is $\rad(P)=P$, shown above. For the inductive step, using part (b) with $I=P^{n-1}$, $J=P$:
    \[
    \rad(P^n) = \rad(P^{n-1}\cdot P) = \rad(P^{n-1})\cap\rad(P) = P \cap P = P,
    \]
    using the induction hypothesis $\rad(P^{n-1})=P$.
    \end{proof}
\end{enumerate}

\item Let $R[X]$ be the polynomial ring over a ring $R$. Prove that the nilradical of $R[X]$ is equal to the Jacobson radical of $R[X]$.

\begin{proof}[Solution]
Since every maximal ideal is prime, intersecting over the (larger) family of prime ideals gives a smaller or equal intersection than intersecting over maximal ideals alone; hence, for any ring, $\caln(R[X]) \subseteq J(R[X])$.

For the reverse containment, we use the identity $\caln(R[X]) = (\caln(R))[X]$ (proof deferred). Take $f(X)\in J(R[X])$. By the standard characterization of the Jacobson radical, $1 - g(X) f(X)$ is a unit for every $g(X) \in R[X]$; taking $g(X) = -X$,
\[
1 + Xf(X) \text{ is a unit in } R[X].
\]
By the Lecture 1 proposition on units of $R[X]$, since the constant term of $1+Xf(X)$ is $1$ (a unit), all remaining coefficients of $1+Xf(X)$ -- i.e.\ all coefficients of $f(X)$ -- are nilpotent. Hence $f(X) \in (\caln(R))[X] = \caln(R[X])$.

Therefore $J(R[X]) \subseteq \caln(R[X])$, and combined with the reverse containment above, $\caln(R[X]) = J(R[X])$.
\end{proof}

\item Let $R[X]$ be the polynomial ring over a ring $R$. Let $J \subset R$ be an ideal. Define
\[
J[X] := \{ a_0 + a_1 X + \cdots + a_n X^n \mid a_i \in J \ \forall\, i \}.
\]
Prove that $J[X]$ is an ideal of $R[X]$. Give an example of an ideal of $R[X]$ which is not of this form. Prove that the rings $R[X]/J[X]$ and $(R/J)[X]$ are isomorphic.

\begin{proof}[Solution]
That $J[X]$ is an ideal of $R[X]$ is a direct verification (closed under addition and absorbs multiplication by $R[X]$, coefficientwise, since $J$ is an ideal of $R$).

\textbf{Example of an ideal not of this form.} Let
\[
I = \{ p(X) \in R[X] \mid \exists\, q(X) \in R[X] \text{ such that } p(X) = X \cdot q(X) \} = \langle X \rangle,
\]
the ideal of polynomials with zero constant term. This is not of the form $J[X]$ for any proper ideal $J \subsetneq R$: an element of $J[X]$ has \emph{every} coefficient in $J$, whereas $I$ places no constraint on coefficients beyond the constant term.

\textbf{Isomorphism.} Define $\varphi: R[X]/J[X] \to (R/J)[X]$ by
\[
\varphi\bigl(r(X) + J[X]\bigr) = \sum_{i=0}^{\deg r} (r_i + J)\, X^i,
\]
i.e.\ reduce each coefficient of $r(X)$ modulo $J$. This is induced by the coefficientwise reduction map $R[X] \to (R/J)[X]$, which is a surjective ring morphism with kernel exactly $J[X]$; by the First Isomorphism Theorem, $\varphi$ is a well-defined ring isomorphism.
\end{proof}

\item A commutative ring is called a \emph{local ring} if it has only one maximal ideal. Let $R$ be a local ring with unique maximal ideal $\mathfrak{m}$. Prove that any element $a \in R \setminus \mathfrak{m}$ is a unit in $R$. Conversely, let $R$ be a ring such that the set of non-units forms an ideal $M$. Prove that $R$ is local with maximal ideal $M$.

\begin{proof}[Solution]
\textbf{($\Rightarrow$)} FTSOC, suppose $a \in R\setminus\mathfrak{m}$ is not a unit. Then $\langle a \rangle \neq R$, i.e.\ $1 \notin \langle a \rangle$. Let
\[
A = \{ I \subseteq R \text{ an ideal} \mid a \in I \text{ but } 1 \notin I \}.
\]
Since $\langle a \rangle \in A$, $A \neq \emptyset$. By Zorn's Lemma, let $M$ be a maximal element of $A$ (ordered by inclusion). One checks $M$ is a maximal ideal of $R$. Since $R$ is local, $M = \mathfrak{m}$, and since $a \in M$, we get $a \in \mathfrak{m}$ -- contradicting $a \notin \mathfrak{m}$. Hence $a$ must be a unit.

\textbf{(Converse)} Suppose $M = \{x \in R \mid x \text{ is not a unit}\}$ is an ideal. Every proper ideal $I$ of $R$ consists entirely of non-units (else $I$ would contain a unit, forcing $I=R$), so $I \subseteq M$. In particular $M$ itself is a proper ideal (as $1 \notin M$), and since every proper ideal is contained in $M$, $M$ is the unique maximal ideal of $R$. Hence $R$ is local with maximal ideal $M$.
\end{proof}

\item An element $e \in R$ is called an \emph{idempotent} if $e^2 = e$. Let $R$ be a local ring. Prove that the only idempotents of $R$ are $0$ and $1$.

\begin{proof}[Solution]
\textbf{Lemma.} If $R$ is a local ring, then for every $a \in R$, either $a$ or $1-a$ is a unit.

\emph{Proof.} Suppose neither $a$ nor $1-a$ is a unit. Then $a \in \mathfrak{m}$ and $1-a \in \mathfrak{m}$ (the unique maximal ideal), so $1 = a + (1-a) \in \mathfrak{m}$, forcing $\mathfrak{m}=R$ -- a contradiction.

Now let $e \in R$ be idempotent: $e^2=e$, i.e.\ $e(e-1)=0$. By the lemma, at least one of $e$, $1-e$ is a unit; multiplying $e(e-1)=0$ by its inverse forces the other factor to vanish. Hence $e=0$ or $e=1$.
\end{proof}
