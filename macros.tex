%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%%%%  BOXED THEOREM ENVIRONMENTS %%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\tcbset{
    thmbox/.style = {
        enhanced, breakable,
        boxrule = 0pt, leftrule = 3pt, sharp corners,
        left = 8pt, right = 8pt, top = 5pt, bottom = 5pt,
        before skip = 12pt, after skip = 12pt,
        attach title to upper, after title = {.\ \ },
        fonttitle = \bfseries, coltitle = black
    }
}

%% Theorem family (green): theorem, proposition, lemma, corollary
\newtcbtheorem[number within=chapter]{theorem}{Theorem}{
    thmbox, colback = subgreen, colframe = maingreen
}{th}

\newtcbtheorem[use counter from=theorem]{proposition}{Proposition}{
    thmbox, colback = subgreen, colframe = maingreen
}{prop}

\newtcbtheorem[use counter from=theorem]{lemma}{Lemma}{
    thmbox, colback = subgreen, colframe = maingreen
}{lem}

\newtcbtheorem[use counter from=theorem]{corollary}{Corollary}{
    thmbox, colback = subgreen, colframe = maingreen
}{cor}

%% Definition family (blue)
\newtcbtheorem[use counter from=theorem]{definition}{Definition}{
    thmbox, colback = sub, colframe = main
}{def}

%% Construction / exercise family (purple)
\newtcbtheorem[use counter from=theorem]{construction}{Construction}{
    thmbox, colback = subpurple, colframe = mainpurple
}{con}

\newtcbtheorem[use counter from=theorem]{exercise}{Exercise}{
    thmbox, colback = subpurple, colframe = mainpurple
}{ex}

%% Remark / observation family (grey, italic title, softer accent)
\newtcbtheorem[use counter from=theorem]{remark}{Remark}{
    thmbox, colback = subgrey, colframe = maingrey,
    fonttitle = \bfseries\itshape
}{rem}

\newtcbtheorem[use counter from=theorem]{observation}{Observation}{
    thmbox, colback = subgrey, colframe = maingrey,
    fonttitle = \bfseries\itshape
}{obs}

\newtcbtheorem[use counter from=theorem]{example}{Example}{
    thmbox, colback = subgrey, colframe = maingrey,
    fonttitle = \bfseries\itshape
}{eg}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%%%% LIGHTWEIGHT (UNBOXED) ENVS %%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Conventions, questions, and notational remarks stay plain --
%% they're asides, not results, and shouldn't compete visually
%% with the boxed theorems/definitions above.

\theoremstyle{remark}
\newtheorem*{notation}{Notation}
\newtheorem*{convention}{Convention}
\newtheorem*{question}{Question}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%%%%   GENERIC COLOR BOXES       %%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Free-standing boxes (not tied to a theorem counter) for
%% highlighting a key point, warning, or side note in the text.

\tcbset{
    sharp corners,
    colback = white,
    before skip = 0.2cm,
    after skip = 0.5cm
}

\newtcolorbox{greenbox}{colback = subgreen,  colframe = maingreen,  boxrule = 0pt, leftrule = 6pt}
\newtcolorbox{purplebox}{colback = subpurple, colframe = mainpurple, boxrule = 0pt, leftrule = 6pt}
\newtcolorbox{greybox}{colback = subgrey,   colframe = maingrey,   boxrule = 0pt, leftrule = 6pt}
\newtcolorbox{blackbox}{colback = black!5,  colframe = mainblack,  boxrule = 0pt, leftrule = 6pt}
